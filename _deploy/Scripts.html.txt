  /* ---------------- tiny DOM helper ---------------- */
  function el(tag, props, kids) {
    var n = document.createElement(tag);
    if (props) {
      for (var k in props) {
        if (!Object.prototype.hasOwnProperty.call(props, k)) continue;
        var v = props[k];
        if (v === null || v === undefined || v === false) continue;
        if (k === 'class') n.className = v;
        else if (k === 'text') n.textContent = v;
        else if (k === 'icon') n.insertAdjacentHTML('afterbegin', v); // constant SVG only
        else if (k.slice(0, 2) === 'on') n.addEventListener(k.slice(2), v);
        else n.setAttribute(k, v === true ? '' : String(v));
      }
    }
    (kids || []).forEach(function (c) {
      if (c === null || c === undefined) return;
      n.appendChild(typeof c === 'string' ? document.createTextNode(c) : c);
    });
    return n;
  }
  function clear(n) { while (n.firstChild) n.removeChild(n.firstChild); }

  var LISTING_ID_RE = /^[1-9][0-9]{4,14}$/;
  var LOCALE_RE = /^[a-z]{2}(?:-[a-z]{2})?$/i;
  function parseEtsy(raw) {
    function fail(code) { return { ok: false, code: code }; }
    if (raw === null || raw === undefined) return fail('EMPTY');
    var s = String(raw).trim();
    if (!s) return fail('EMPTY');
    if (s.length > 2048) return fail('TOO_LONG');
    if (/[\s\u0000-\u001f\u007f\\]/.test(s)) return fail('MALFORMED');
    var rest, m = /^([a-z][a-z0-9+.\-]*):\/\//i.exec(s);
    if (m) {
      var sc = m[1].toLowerCase();
      if (sc !== 'https' && sc !== 'http') return fail('SCHEME');
      rest = s.slice(m[0].length);
    } else if (/^[a-z][a-z0-9+.\-]*:/i.test(s)) { return fail('SCHEME'); }
    else { rest = s; }
    var cut = rest.search(/[\/?#]/);
    var auth = cut === -1 ? rest : rest.slice(0, cut);
    var tail = cut === -1 ? '' : rest.slice(cut);
    if (!auth) return fail('MALFORMED');
    if (auth.indexOf('@') !== -1) return fail('HOST');
    var hm = /^([a-z0-9.\-]+)(?::(\d{1,5}))?$/i.exec(auth);
    if (!hm) return fail('MALFORMED');
    if (hm[2] !== undefined && hm[2] !== '443' && hm[2] !== '80') return fail('HOST');
    if (ALLOWED_HOSTS.indexOf(hm[1].toLowerCase()) === -1) return fail('HOST');
    var path = tail.split(/[?#]/)[0];
    if (path.charAt(0) !== '/') return fail('NOT_LISTING');
    var segs = path.split('/').filter(function (x) { return x.length > 0; });
    if (segs.length && LOCALE_RE.test(segs[0])) segs.shift();
    if (!segs.length || segs[0].toLowerCase() !== 'listing') return fail('NOT_LISTING');
    if (segs.length < 2) return fail('BAD_ID');
    if (segs.length > 3) return fail('NOT_LISTING');
    if (!LISTING_ID_RE.test(segs[1])) return fail('BAD_ID');
    return { ok: true, listingId: segs[1] };
  }
  window.__parseEtsyClient = parseEtsy; // exposed for automated tests only

  /* ---------------- server calls ---------------- */
  function rpc(req) {
    return new Promise(function (resolve) {
      try {
        google.script.run
          .withSuccessHandler(function (r) { resolve(r || { ok: false, code: 'SERVER', message: 'No response from the server.' }); })
          .withFailureHandler(function () { resolve({ ok: false, code: 'NETWORK', message: 'Could not reach the server. Check your connection and try again.' }); })
          .api(S.token, req);
      } catch (e) {
        resolve({ ok: false, code: 'NETWORK', message: 'Could not reach the server. Check your connection and try again.' });
      }
    });
  }

  function call(action, payload, quiet) {
    if (!quiet) setSaving(1);
    return rpc({ action: action, storeId: S.storeId, payload: payload || {} }).then(function (r) {
      if (!quiet) setSaving(-1);
      return r;
    });
  }

  /** Account-level actions (no store in the request): sign out, store list, create store. */
  function callGlobal(action, payload, quiet) {
    if (!quiet) setSaving(1);
    return rpc({ action: action, payload: payload || {} }).then(function (r) {
      if (!quiet) setSaving(-1);
      return r;
    });
  }

  function setSaving(d) {
    S.inflight = Math.max(0, S.inflight + d);

  function fail(r) {
    if (r.code === 'AUTH_REQUIRED') { gateSignIn(S.token ? 'Your session has ended. Please sign in again.' : r.message); return; }
    if (r.code === 'DENIED') { showGate('Access denied', r.message, 'denied'); return; }
    toast(r.message || 'Something went wrong. Please try again.', true);
  }

  function showGate(title, text, mode) {

    $('gateLink').hidden = (m !== 'signin' && m !== 'denied') || (m === 'denied' && !canLogin);
    $('gateLink').textContent = m === 'denied' ? 'Use another Google account' : 'Sign in with Google';

  function gateSignIn(msg) {
    if (BOOT.authMode === 'SESSION' && BOOT.loginEnabled) {
      showGate('Sign in to continue', S.notice || msg || 'Use your Google account to open this tracker.', true);
    } else if (BOOT.authMode === 'SESSION') {
      showGate('Sign-in unavailable', 'Sign-in is not configured for this tracker yet. Please contact the store owner.');

  /* ---- browser-bound sign-in: the secret verifier never leaves this browser ---- */
  var VKEY = 'ls_login_verifier';
  function b64url(buf) {
    var s = '', a = new Uint8Array(buf);
    for (var i = 0; i < a.length; i++) s += String.fromCharCode(a[i]);
    return btoa(s).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
  }
  function storeVerifier(v) {
    try { localStorage.setItem(VKEY, v); return true; } catch (e) { /* try next */ }
    try { sessionStorage.setItem(VKEY, v); return true; } catch (e2) { return false; }
  }
  function readVerifier() {
    try { var v = localStorage.getItem(VKEY); if (v) return v; } catch (e) { /* try next */ }
    try { return sessionStorage.getItem(VKEY) || ''; } catch (e2) { return ''; }
  }
  function clearVerifier() {
    try { localStorage.removeItem(VKEY); } catch (e) { /* ignore */ }
    try { sessionStorage.removeItem(VKEY); } catch (e2) { /* ignore */ }
  }
  function beginLogin() {
    var btn = $('gateLink');
    if (!(window.crypto && crypto.subtle && crypto.getRandomValues)) {
      showGate('Sign-in unavailable', 'This browser cannot sign in securely. Please use a current version of Chrome, Edge, Safari or Firefox.');
      return;
    }
    var rnd = new Uint8Array(32); crypto.getRandomValues(rnd);
    var verifier = b64url(rnd);
    if (!storeVerifier(verifier)) {
      showGate('Sign-in unavailable', 'Your browser is blocking site storage, which is needed to sign in securely. Allow storage for this page and try again.');
      return;
    }
    btn.disabled = true;
    crypto.subtle.digest('SHA-256', new TextEncoder().encode(verifier)).then(function (d) {
      return call('startLogin', { challenge: b64url(d) }, true);
    }).then(function (r) {
      if (!r.ok) { btn.disabled = false; clearVerifier(); toast(r.message || 'Sign-in is not available right now.', true); return; }
      try { window.top.location.href = r.data.authUrl; } catch (e) { window.open(r.data.authUrl, '_top'); }
    });
  }
  /** Ends this person's sessions on the server (all browsers) and in this page. */
  function signOut() {
    var b = $('gateSignOut'); b.disabled = true;
    return callGlobal('signOut', {}, true).then(function () {
      S.token = '';
      try { sessionStorage.removeItem('ls_tok_' + BOOT.storeId); } catch (e) { /* ignore */ }
      clearVerifier();
      firstLoaded = false;
      S.notice = '';
      showGate('Signed out', 'You have been signed out of Listing Studio.', BOOT.authMode === 'SESSION' && BOOT.loginEnabled ? 'signin' : 'none');
    });
  }
  function tryAgain() {
    var b = $('gateRetry'); b.disabled = true;
    loadState(true).then(function (okay) { if (!okay) b.disabled = false; });
  }

  function redeemPending() {
    var v = readVerifier();
    if (!v) { S.notice = 'Sign-in could not be completed in this browser. Please sign in again.'; gateSignIn(); return; }
    call('redeemLogin', { pending: BOOT.pending, verifier: v }, true).then(function (r) {
      clearVerifier();
      if (!r.ok) { S.notice = r.message || 'Sign-in could not be completed. Please sign in again.'; gateSignIn(); return; }
      S.token = r.data.token;
      try { sessionStorage.setItem('ls_tok_' + BOOT.storeId, S.token); } catch (e) { /* storage unavailable: fine */ }
      loadState(true);
    });
  }

  var toastTimer;
  function toast(msg, isError) {
    var t = $('toast');
    t.textContent = msg;
    t.className = 'toast' + (isError ? ' error' : '');
    t.hidden = false;
    clearTimeout(toastTimer);
    toastTimer = setTimeout(function () { t.hidden = true; }, isError ? 6500 : 3500);
  }

  function confirmDialog(title, text, okLabel, neutral) {
    return new Promise(function (resolve) {
      var m = $('modal');
      $('modalTitle').textContent = title;
      $('modalText').textContent = text;
      $('modalOk').textContent = okLabel || 'Confirm';
      $('modalOk').className = 'btn ' + (neutral ? 'btn-primary' : 'btn-danger');
      m.hidden = false;
      var prev = document.activeElement;
      function done(v) {
        m.hidden = true;
        $('modalOk').removeEventListener('click', ok);
        $('modalCancel').removeEventListener('click', no);
        document.removeEventListener('keydown', key, true);
        if (prev && prev.focus) { try { prev.focus(); } catch (e) { /* element gone */ } }
        resolve(v);
      }
      function ok() { done(true); }
      function no() { done(false); }
      function key(e) { if (e.key === 'Escape') { e.preventDefault(); done(false); } }
      $('modalOk').addEventListener('click', ok);
      $('modalCancel').addEventListener('click', no);
      document.addEventListener('keydown', key, true);
      $('modalCancel').focus();
    });
  }

  /* ---------------- state ---------------- */
  function recount() {
    var c = { total: 0, completed: 0, remaining: 0, errors: 0 };
    S.records.forEach(function (r) {
      if (r.status === 'VALID') { c.total++; if (r.done) c.completed++; else c.remaining++; }
      else c.errors++;
    });
    S.counts = c;
  }

  function setState(d) {
    S.records = d.records || [];
    S.role = d.role;
    S.email = d.email;
    S.storeName = d.store.name;
    if (d.store.id) S.storeId = d.store.id;
    S.caps = d.caps || {};
    S.stores = d.stores || [];
    S.sheetUrl = d.sheetUrl || '';
    S.rev = d.rev;
    // drop UI state for records that no longer exist
    var ids = {};
    S.records.forEach(function (r) { ids[r.id] = true; });
    [S.expanded, S.drafts, S.pending].forEach(function (o) { Object.keys(o).forEach(function (k) { if (!ids[k]) delete o[k]; }); });
    if (S.editing && !ids[S.editing]) S.editing = '';
    if (S.filter === 'errors' && !isAdmin()) S.filter = 'all';
    recount();
    render();
    renderSwitcher();
  }

  function loadState(initial) {
    return call('getState', {}, !initial).then(function (r) {
      if (!r.ok) { if (initial) fail(r); return false; }
      if (initial) onFirstLoad(true);
      setState(r.data);
      return true;
    });
  }

  var firstLoaded = false;
  function onFirstLoad(force) {
    if (firstLoaded && !force) return;
    firstLoaded = true;
    $('gate').hidden = true;
    $('app').hidden = false;
  }

  /* ---------------- derived views ---------------- */
  function visible() {
    var q = S.query.trim().toLowerCase();
    return S.records.filter(function (r) {
      var err = r.status !== 'VALID';
      if (S.filter === 'remaining' && (err || r.done)) return false;
      if (S.filter === 'done' && (err || !r.done)) return false;
      if (S.filter === 'errors' && !err) return false;
      if (q) {
        var hay = (r.listingId + ' ' + r.url + ' ' + (r.notes || '')).toLowerCase();
        if (hay.indexOf(q) === -1) return false;
      }
      return true;
    });
  }

  function errors() { return S.records.filter(function (r) { return r.status !== 'VALID'; }); }

  /* ---------------- render ---------------- */
  var appliedAdmin = null;
  var detached = [];
  function applyRole() {
    var admin = isAdmin();
    if (appliedAdmin !== admin) {
      appliedAdmin = admin;
      if (!admin) {
        // Team view: administrative controls are REMOVED from the DOM, not merely hidden.
        // (A marker is left behind so switching to a store where this person is an
        // administrator can put the very same controls back.)
        Array.prototype.slice.call(document.querySelectorAll('.admin-only')).forEach(function (n) {
          var marker = document.createComment('admin-only');
          n.parentNode.replaceChild(marker, n);
          detached.push({ node: n, marker: marker });
        });
      } else {
        detached.forEach(function (d) { if (d.marker.parentNode) d.marker.parentNode.replaceChild(d.node, d.marker); });
        detached = [];
        bindAdminControls();
      }
    }
    $('accessLabel').textContent = admin ? 'Administrator access' : 'Team access';
  }

  function render() {
    var y = window.pageYOffset || 0;
    applyRole();
    renderHeader();
    renderKpis();
    renderProgress();
    renderErrorPanel();
    renderFilters();
    renderTable();
    window.scrollTo(0, y);
  }

  function renderHeader() {
    $('storeName').textContent = S.storeName;
    document.title = S.storeName + ' · Listing Studio';
  }

  function renderKpis() {
    var box = $('kpis');
    clear(box);
    box.className = 'kpis' + (isAdmin() ? '' : ' three');
    var defs = [
      { key: 'total', label: 'Total listings', icon: ICON.total, filter: 'all', val: S.counts.total },
      { key: 'completed', label: 'Completed', icon: ICON.done, filter: 'done', val: S.counts.completed, cls: 'good' },
      { key: 'remaining', label: 'Remaining', icon: ICON.remaining, filter: 'remaining', val: S.counts.remaining }
    ];
    if (isAdmin()) defs.push({ key: 'errors', label: 'Errors', icon: ICON.errors, filter: 'errors', val: S.counts.errors, cls: S.counts.errors > 0 ? 'bad' : '' });
    defs.forEach(function (d) {
      var b = el('button', {
        type: 'button', class: 'kpi' + (S.filter === d.filter && S.filter !== 'all' ? ' active' : ''), 'data-kpi': d.key,

        onclick: function () { S.filter = d.filter; renderKpis(); renderFilters(); renderTable(); }
      }, [
        el('div', { class: 'kpi-top' }, [el('span', { icon: d.icon }), el('span', { text: d.label })]),
        el('div', { class: 'kpi-value ' + (d.cls || ''), text: String(d.val) })
      ]);
      box.appendChild(b);
    });
  }

  function renderProgress() {
    var c = S.counts;
    var pct = c.total > 0 ? Math.min(100, Math.max(0, Math.round((c.completed / c.total) * 100))) : 0;
    $('pct').textContent = pct + '%';
    $('barFill').style.width = pct + '%';
    $('bar').setAttribute('aria-valuenow', String(pct));
    $('progressText').textContent = c.completed + ' of ' + c.total + ' unique listings completed';
  }

  function renderFilters() {
    var box = $('filters');
    clear(box);
    var defs = [['all', 'All'], ['remaining', 'Remaining'], ['done', 'Done']];
    if (isAdmin()) defs.push(['errors', 'Errors']);
    defs.forEach(function (d) {
      box.appendChild(el('button', {
        type: 'button', class: 'seg', 'data-filter': d[0], 'aria-pressed': S.filter === d[0] ? 'true' : 'false', text: d[1],
        onclick: function () { S.filter = d[0]; renderKpis(); renderFilters(); renderTable(); }
      }));
    });
  }

  function renderErrorPanel() {
    var p = $('errorPanel');
    if (!p) return;
    var errs = errors();
    if (!errs.length) { p.hidden = true; clear(p); return; }
    var cur = null;
    errs.forEach(function (r) { if (r.id === S.focusId) cur = r; });
    if (!cur) cur = errs[errs.length - 1];
    var e = cur.error || {};
    clear(p);
    var body = el('div', { class: 'error-body' }, [
      el('h2', { class: 'error-head', text: e.heading || 'Problem with this entry' }),
      el('p', { class: 'error-msg', text: e.message || '' })
    ]);
    if (e.kind === 'duplicate') body.appendChild(el('p', { class: 'error-meta', text: 'Original listing: ' + e.originalState }));
    var actions = el('div', { class: 'error-actions' });
    if (e.kind === 'duplicate') {
      actions.appendChild(el('button', { type: 'button', class: 'btn btn-sm btn-err', 'data-act': 'find', text: 'Find original listing', onclick: function () { findOriginal(e.originalId); } }));
    }
    actions.appendChild(el('button', { type: 'button', class: 'btn btn-sm btn-err', 'data-act': 'clear', text: 'Clear invalid entry', onclick: function () { clearEntry(cur); } }));
    body.appendChild(actions);
    if (errs.length > 1) {
      var others = errs.length - 1;
      body.appendChild(el('p', { class: 'error-more' }, [
        others + (others === 1 ? ' other entry needs' : ' other entries need') + ' attention. ',
        el('button', { type: 'button', class: 'linklike', text: 'View all errors', onclick: function () { S.filter = 'errors'; renderKpis(); renderFilters(); renderTable(); } })
      ]));
    }
    p.appendChild(el('span', { class: 'error-icon', icon: ICON.alert }));
    p.appendChild(body);
    p.hidden = false;
  }

  function findOriginal(id) {
    S.filter = 'all'; S.query = ''; $('search').value = '';
    S.flashId = id;
    renderKpis(); renderFilters(); renderTable();
    var row = document.querySelector('tr[data-id="' + id + '"]');
    if (row && row.scrollIntoView) row.scrollIntoView({ block: 'center', behavior: 'smooth' });
  }

  /* ---------------- table ---------------- */
  function thead() {
    var h = $('thead');
    clear(h);
    var cols = [['#', 'col-n c'], ['Etsy Listing Link', 'col-link'], ['Listing ID', 'col-id'], ['Open', 'col-open c'], ['Notes', 'col-notes'], ['Images Done', 'col-done c']];
    if (isAdmin() && S.showTs) { cols.push(['Date Added', 'col-ts']); cols.push(['Date Completed', 'col-ts']); }
    var tr = el('tr');
    cols.forEach(function (c) { tr.appendChild(el('th', { scope: 'col', class: c[1], text: c[0] })); });
    h.appendChild(tr);
    return cols.length;
  }

  function renderTable() {
    var ncols = thead();
    var body = $('tbody');
    clear(body);
    var rows = visible();
    rows.forEach(function (r) {
      body.appendChild(r.status === 'VALID' ? validRow(r) : errorRow(r));
      if (S.expanded[r.id] && r.status === 'VALID') body.appendChild(notesRow(r, ncols));
    });
    var showBlank = isAdmin() && S.filter === 'all' && !S.query.trim();
    if (showBlank) for (var i = 0; i < S.extraRows; i++) body.appendChild(blankRow(i));

    var msg = $('emptyMsg');
    if (!rows.length && !showBlank) {
      msg.hidden = false;
      msg.textContent = S.records.length ? 'No listings match this view.' : 'No listings yet.';
    } else if (!rows.length && showBlank && !S.records.length) {
      msg.hidden = false;
      msg.textContent = 'No listings yet. Paste an Etsy link into the first row, or use “Add listings”.';
    } else {
      msg.hidden = true;
    }
    if (S.flashId) {
      var f = body.querySelector('tr[data-id="' + S.flashId + '"]');
      if (f) f.classList.add('flash');
      S.flashId = '';

  /* ---------------- actions ---------------- */
  function onToggle(r, btn) {
    if (S.pending[r.id]) return;
    S.pending[r.id] = true;
    btn.disabled = true;
    btn.setAttribute('aria-busy', 'true');
    var want = !r.done;
    call('setDone', { recordId: r.id, done: want }).then(function (res) {
      delete S.pending[r.id];
      if (res.ok) {
        var rec = res.data.record;
        S.records = S.records.map(function (x) { return x.id === rec.id ? Object.assign({}, x, rec) : x; });
        if (res.data.rev === S.rev + 1) S.rev = res.data.rev;
        recount();
        render();
      } else {
        fail(res);
        if (res.code === 'NOT_FOUND' || res.code === 'NOT_ALLOWED') loadState(false); else render();
      }
    });
  }

  function saveNote(r, text) {
    call('saveNote', { recordId: r.id, note: text, expectedVersion: r.version }).then(function (res) {
      if (res.ok) {
        var rec = res.data.record;
        S.records = S.records.map(function (x) { return x.id === rec.id ? Object.assign({}, x, rec) : x; });
        delete S.drafts[r.id];
        delete S.expanded[r.id];
        if (res.data.rev === S.rev + 1) S.rev = res.data.rev;
        render();
        toast('Note saved.');
      } else {
        fail(res);
        if (res.code === 'CONFLICT' || res.code === 'NOT_FOUND') loadState(false);
      }
    });
  }

  function submitUrlEdit(r, value, confirmed) {
    var v = String(value).trim();
    if (!v || v === r.url) { S.editing = ''; renderTable(); return Promise.resolve(); }
    var payload = { recordId: r.id, url: v, expectedVersion: r.version };
    if (confirmed) payload.confirm = true;
    return call('updateUrl', payload).then(function (res) {
      if (res.ok) {
        S.editing = '';
        setState(res.data.state);
        var rec = S.records.filter(function (x) { return x.id === res.data.focusId; })[0];
        if (rec && rec.status !== 'VALID') S.focusId = rec.id;
        renderErrorPanel();
        return;
      }
      if (res.code === 'CONFIRM_REQUIRED') {
        return confirmDialog('Replace this listing?', res.message, 'Replace listing').then(function (yes) {
          if (yes) return submitUrlEdit(r, v, true);
          S.editing = ''; renderTable();
        });
      }
      fail(res);
      if (res.code === 'CONFLICT' || res.code === 'NOT_FOUND') loadState(false);
    });
  }

  function addUrls(lines) {
    return call('addListings', { urls: lines }).then(function (res) {
      if (!res.ok) { fail(res); return false; }
      var results = res.data.results;
      setState(res.data.state);
      var bad = results.filter(function (x) { return x.status !== 'VALID'; });
      var good = results.length - bad.length;
      if (bad.length) { S.focusId = bad[bad.length - 1].id; renderErrorPanel(); }
      if (results.length > 1 || (good && !bad.length)) {
        toast(good + (good === 1 ? ' listing added.' : ' listings added.') + (bad.length ? ' ' + bad.length + ' need attention.' : ''));
      }
      return true;
    });
  }

  function submitBlank(inp, lines) {
    if (inp.dataset.busy) return;
    var clean = lines.map(function (l) { return l.trim(); }).filter(Boolean);
    if (!clean.length) return;
    inp.dataset.busy = '1';
    inp.value = clean[0];
    inp.disabled = true;
    addUrls(clean).then(function (ok) { if (!ok) { inp.disabled = false; delete inp.dataset.busy; } });
  }

      else { fail(res); if (res.code === 'NOT_FOUND') loadState(false); }

  }

  function copyText(text, input) {
    function fallback() {
      try {
        var t = input || el('textarea', { 'aria-hidden': 'true', style: 'position:fixed;left:-9999px;top:0' });
        if (!input) { t.value = text; document.body.appendChild(t); }
        t.focus(); t.select();
        var okay = document.execCommand && document.execCommand('copy');
        if (!input) document.body.removeChild(t);
        return !!okay;
      } catch (e) { return false; }
    }
    function done(okay) { toast(okay ? 'Link copied.' : 'Select the link and press Ctrl+C to copy it.', !okay); }
    try {
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(text).then(function () { done(true); }, function () { done(fallback()); });
        return;
      }
    } catch (e) { /* use fallback */ }
    done(fallback());
  }

    ).then(function (yes) {
      if (!yes) { cb.checked = !want; return; }
      mutateTeam('setMemberRole', { email: m.email, role: 'admin', manager: want, confirm: true }, 'Permission updated.');
    });
  }

  function toggleStatus(m) {
    var disabling = m.status === 'active';
    var payload = { email: m.email, status: disabling ? 'disabled' : 'active' };
    var go = function (confirmed) {
      if (confirmed) payload.confirm = true;
      return mutateTeam('setMemberStatus', payload, disabling ? 'Access disabled.' : 'Access restored.');
    };
    if (m.role === 'admin') {
      return confirmDialog(
        (disabling ? 'Disable ' : 'Restore ') + m.email + '?',
        disabling ? 'They are an administrator. They will be blocked immediately, even if signed in now. You can restore access later.' : 'They will be an administrator of this store again.',
        disabling ? 'Disable access' : 'Restore access', !disabling
      ).then(function (yes) { if (yes) return go(true); });
    }
    return go(false);
  }

      'Remove'
    ).then(function (yes) {
      if (yes) return mutateTeam('removeMember', { email: m.email, confirm: true }, 'Person removed.');
    });
  }

  /* ---------------- Create new store ---------------- */
  function newRequestId() {
    var a = new Uint8Array(16);
    crypto.getRandomValues(a);
    return Array.prototype.map.call(a, function (b) { return ('0' + b.toString(16)).slice(-2); }).join('');
  }
  function splitEmails(text) { return String(text || '').split(/[\s,;]+/).filter(Boolean); }
  var EMAIL_UX_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

  function createFeedback(msg, good) {
    var f = $('createFeedback');
    f.textContent = msg || '';
    f.className = 'form-feedback' + (good ? ' good' : '');
  }

  function resetCreate() {
    ST.create = { reqId: newRequestId(), failed: false, busy: false, done: null };
    $('newStoreName').value = ''; $('newStoreAdmins').value = ''; $('newStoreTeam').value = '';
    createFeedback('');
    $('createSubmit').disabled = false; $('createSubmit').textContent = 'Create Store';
    $('createForm').hidden = false; $('createDone').hidden = true;
  }

  function prepareCreate() {
    if (!ST.create.reqId) ST.create.reqId = newRequestId();
    $('createForm').hidden = !!ST.create.done;
    $('createDone').hidden = !ST.create.done;
    if (!ST.create.done) $('newStoreName').focus();
  }

  function refreshStores() {
    return callGlobal('listMyStores', {}, true).then(function (r) { if (r.ok) { S.stores = r.data.stores; renderSwitcher(); } });
  }

  function submitCreate() {
    if (ST.create.busy) return;
    var name = $('newStoreName').value.trim();
    var admins = splitEmails($('newStoreAdmins').value);
    var team = splitEmails($('newStoreTeam').value);
    if (!name) { createFeedback('Enter a store name.'); $('newStoreName').focus(); return; }
    var all = admins.concat(team);
    for (var i = 0; i < all.length; i++) if (!EMAIL_UX_RE.test(all[i])) { createFeedback('“' + all[i] + '” is not a valid email address.'); return; }
    var lower = {};
    for (var j = 0; j < admins.length; j++) lower[admins[j].toLowerCase()] = true;
    for (var k = 0; k < team.length; k++) if (lower[team[k].toLowerCase()]) { createFeedback(team[k] + ' is in both lists. Choose one.'); return; }
    ST.create.busy = true;
    $('createSubmit').disabled = true;
    $('createSubmit').textContent = 'Creating…';
    createFeedback('Creating the store and its Google Sheet. This takes about 20 seconds. Please keep this page open.', true);
    callGlobal('createStore', { requestId: ST.create.reqId, name: name, admins: admins, team: team }).then(function (r) {
      ST.create.busy = false;
      $('createSubmit').disabled = false;
      $('createSubmit').textContent = 'Create Store';
      if (r.ok) {
        ST.create.done = r.data;
        createFeedback('');
        renderCreated();
        refreshStores();
        toast('Store created.');
        return;
      }
      // Network problems, BUSY and half-finished set-ups keep the SAME request id: pressing the
      // button again continues the same store and can never create a second one.
      ST.create.failed = r.code === 'SERVER' || r.code === 'NETWORK' || r.code === 'BUSY';
      createFeedback(r.message || 'Something went wrong. Please try again.');
      if (r.code === 'AUTH_REQUIRED' || r.code === 'DENIED') fail(r);
    });
  }

  function renderCreated() {
    var d = ST.create.done;
    if (!d) return;
    $('createForm').hidden = true;
    $('createDone').hidden = false;
    $('createdName').textContent = d.name;
    $('createdLink').value = d.appUrl || '';
    var ok = !!d.sheetUrl && SHEET_URL_RE.test(d.sheetUrl);
    ['createdSheet', 'createdSheetBtn'].forEach(function (id) {
      var n = $(id);
      n.hidden = !ok;
      if (ok) n.setAttribute('href', d.sheetUrl);
    });
    var w = $('createdWarn');
    clear(w);
    (d.warnings || []).forEach(function (t) { w.appendChild(el('li', { text: t })); });
    w.hidden = !(d.warnings || []).length;
  }

  function renderSwitcher() {
    var wrap = $('switchWrap');

  function resetStoreView() {
    S.filter = 'all'; S.query = ''; S.showTsDraft = false;
    S.expanded = {}; S.drafts = {}; S.pending = {}; S.editing = ''; S.focusId = ''; S.flashId = '';
    S.extraRows = 2;
    var q = $('search'); if (q) q.value = '';

  function switchStore(id) {
    if (!id || id === S.storeId) { renderSwitcher(); return Promise.resolve(); }
    setSaving(1);
    return rpc({ action: 'getState', storeId: id, payload: {} }).then(function (r) {
      setSaving(-1);
      if (!r.ok) {
        renderSwitcher();   // snap the selector back to the store that is still open
        if (r.code === 'AUTH_REQUIRED') { fail(r); return; }
        toast(r.code === 'DENIED' ? 'You no longer have access to that store.' : (r.message || 'Could not open that store.'), true);
        refreshStores();
        return;
      }
      resetStoreView();
      S.storeId = id;
      setState(r.data);
      toast('Opened ' + r.data.store.name + '.');
    });
  }

  function bulkLines() {
    return $('bulkText').value.split(/\r?\n/).map(function (l) { return l.trim(); }).filter(Boolean);
  }

  function analyzeBulk() {
    var box = $('bulkFeedback');
    if (!box) return;
    var lines = bulkLines();
    clear(box);
    $('bulkSubmit').disabled = !lines.length || lines.length > MAX_BATCH;
    if (!lines.length) return;
    var known = {};
    S.records.forEach(function (r) { if (r.status === 'VALID') known[r.listingId] = r.n; });
    var seen = {};
    var nNew = 0, nDup = 0, nBad = 0;
    var issues = [];
    lines.forEach(function (l, i) {
      var p = parseEtsy(l);
      if (!p.ok) { nBad++; issues.push('Line ' + (i + 1) + ': not a valid Etsy product listing link'); return; }
      if (known[p.listingId]) { nDup++; issues.push('Line ' + (i + 1) + ': duplicate — ID ' + p.listingId + ' already exists in row ' + known[p.listingId]); return; }
      if (seen[p.listingId]) { nDup++; issues.push('Line ' + (i + 1) + ': duplicate of line ' + seen[p.listingId] + ' in this paste'); return; }
      seen[p.listingId] = i + 1;
      nNew++;
    });
    var sum = nNew + ' new · ' + nDup + ' duplicate · ' + nBad + ' invalid';
    box.appendChild(el('span', { class: 'sum', text: lines.length > MAX_BATCH ? 'Too many links (max ' + MAX_BATCH + ' at a time).' : sum }));
    if (issues.length) {
      var ul = el('ul');
      issues.slice(0, 20).forEach(function (t) { ul.appendChild(el('li', { text: t })); });
      if (issues.length > 20) ul.appendChild(el('li', { text: '…and ' + (issues.length - 20) + ' more' }));
      box.appendChild(ul);
    }
  }

  /* ---------------- polling (periodic, NOT real-time) ---------------- */
  function poll() {
    if (document.hidden || S.inflight > 0 || Object.keys(S.pending).length || !firstLoaded) return;
    call('getRevision', {}, true).then(function (r) {
      if (!r.ok) { if (r.code === 'AUTH_REQUIRED' || r.code === 'DENIED') fail(r); return; }
      if (r.data.name && r.data.name !== S.storeName) { S.storeName = r.data.name; renderHeader(); }
      if (r.data.rev === S.rev) return;
      var a = document.activeElement;
      if (a && a.closest && a.closest('#tbody') && (a.tagName === 'TEXTAREA' || a.tagName === 'INPUT')) return; // never clobber typing; next tick retries
      loadState(false);
    });
  }

    $('search').addEventListener('input', debounce(function () { S.query = $('search').value; renderTable(); }, 120));
    document.addEventListener('visibilitychange', function () { if (!document.hidden) poll(); });
    setInterval(poll, POLL_MS);

    $('gateLink').addEventListener('click', beginLogin);
    $('gateRetry').addEventListener('click', tryAgain);
    $('gateSignOut').addEventListener('click', signOut);
    $('storeSwitch').addEventListener('change', function () { switchStore($('storeSwitch').value); });
    var key = 'ls_tok_' + BOOT.storeId;
    try { S.token = sessionStorage.getItem(key) || ''; } catch (e) { S.token = ''; }

    if (!BOOT.storeId) {
      showGate('Missing store link', 'Please open the link the store owner sent you. It ends with ?store=\u2026');
      return;
    }
    if (BOOT.pending) { redeemPending(); return; }
    if (BOOT.authMode === 'SESSION' && !S.token) { gateSignIn(); return; }
    loadState(true);
  }

  window.__LS = { S: S, render: render, poll: poll };
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init); else init();
})();
