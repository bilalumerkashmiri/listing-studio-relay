/**
 * Setup.gs — one-time project setup, master template and new-store registration.
 * Run these functions from the Apps Script editor (Run button) as the OWNER.
 * Everything here is idempotent: running it twice never duplicates headers,
 * triggers, formatting rules, protections or records.
 */

/* ---- Edit these, then run registerNewStore() ---- */
var NEW_STORE = {
  spreadsheetId: '',                 // ID of the COPY of the master template (from its URL)
  name: 'My Etsy Store',
  admins: ['owner@example.com'],     // Google accounts that manage listings
  team: ['designer@example.com'],    // Google accounts that only tick "Images Done"
  existingData: ''                   // only needed if the copy already holds rows: 'EMPTY' or 'PRESERVE'
};

/* ---- Edit these, then run updateStoreAccess() ---- */
var ACCESS_UPDATE = {
  storeId: '',
  admins: [],
  team: []
};

/**
 * SECURITY: every top-level function whose name does not end in "_" can be invoked by ANY
 * visitor of the web app through google.script.run, because the app runs as the owner.
 * The functions below exist only to be run by the owner from the Apps Script editor, so each
 * one starts with assertOwnerCaller_(): the caller must be the verified owner. An anonymous or
 * signed-in-but-not-owner web visitor gets an empty / different active-user email and is refused
 * before anything is read or changed.
 */
function assertOwnerCaller_() {
  var caller = '';
  try { caller = String(Session.getActiveUser().getEmail() || '').toLowerCase(); } catch (e) { caller = ''; }
  var owner = scriptProp_('OWNER_EMAIL').toLowerCase();
  var effective = '';
  try { effective = String(Session.getEffectiveUser().getEmail() || '').toLowerCase(); } catch (e2) { effective = ''; }
  var expected = owner || effective; // first run: OWNER_EMAIL not recorded yet
  if (!caller || !expected || caller !== expected) {
    throw new Error('Not authorized.');
  }
}

var PROTECT_DESC_ = 'Listing Studio: automatic fields';
var EMAIL_RE_ = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;

/** Step 1 (once). Creates the signing key and records the owner. Safe to re-run. */
function initializeProject() {
  assertOwnerCaller_();
  var props = PropertiesService.getScriptProperties();
  if (!props.getProperty('SIGNING_KEY')) {
    props.setProperty('SIGNING_KEY', Utilities.getUuid() + Utilities.getUuid() + Utilities.getUuid());
  }
  if (!props.getProperty('OWNER_EMAIL')) {
    props.setProperty('OWNER_EMAIL', Session.getEffectiveUser().getEmail());
  }
  if (!props.getProperty('AUTH_MODE')) props.setProperty('AUTH_MODE', 'SESSION');
  Logger.log('Initialized. Next: add OAUTH_CLIENT_ID, OAUTH_CLIENT_SECRET and WEBAPP_URL under Project Settings > Script properties, then run checkConfiguration().');
  return checkConfiguration();
}

/** Invalidates every signed-in session immediately. */
function rotateSigningKey() {
  assertOwnerCaller_();
  PropertiesService.getScriptProperties().setProperty('SIGNING_KEY', Utilities.getUuid() + Utilities.getUuid() + Utilities.getUuid());
  Logger.log('Signing key rotated. All users must sign in again.');
}

/** Reports configuration status without printing secrets. */
function checkConfiguration() {
  assertOwnerCaller_();
  var mode = getAuthMode_();
  var report = {
    authMode: mode,
    signingKey: scriptProp_('SIGNING_KEY').length >= 32 ? 'ok' : 'MISSING (run initializeProject)',
    ownerEmail: scriptProp_('OWNER_EMAIL') || 'MISSING (run initializeProject)',
    oauthClientId: scriptProp_('OAUTH_CLIENT_ID') ? 'set' : (mode === 'SESSION' ? 'MISSING' : 'not needed'),
    oauthClientSecret: scriptProp_('OAUTH_CLIENT_SECRET') ? 'set' : (mode === 'SESSION' ? 'MISSING' : 'not needed'),
    webappUrl: scriptProp_('WEBAPP_URL') || (mode === 'SESSION' ? 'MISSING' : 'not needed'),
    oauthRedirectUri: scriptProp_('OAUTH_REDIRECT_URI') || '(not set: the web app URL is used)',
    masterTemplate: scriptProp_('MASTER_TEMPLATE_ID') ? 'set' : 'MISSING (run createMasterTemplate)',
    triggersUsed: ScriptApp.getProjectTriggers().length + ' of ' + MAX_TRIGGERS_,
    stores: listStores_().map(function (s) {
      return {
        id: s.id, name: s.name, spreadsheetId: s.spreadsheetId, admins: s.admins.length, team: s.team.length,
        url: scriptProp_('WEBAPP_URL') ? scriptProp_('WEBAPP_URL') + '?store=' + s.id : '(set WEBAPP_URL first)'
      };
    })
  };
  Logger.log(JSON.stringify(report, null, 2));
  return report;
}

/** Creates (or repairs) "Listing Studio — Master Template": clean, unregistered, zero rows. */
function createMasterTemplate() {
  assertOwnerCaller_();
  var props = PropertiesService.getScriptProperties();
  var ss = null;
  var existingId = props.getProperty('MASTER_TEMPLATE_ID');
  if (existingId) {
    try { ss = SpreadsheetApp.openById(existingId); } catch (e) { ss = null; }
  }
  if (!ss) {
    ss = SpreadsheetApp.create('Listing Studio — Master Template');
    props.setProperty('MASTER_TEMPLATE_ID', ss.getId());
  }
  prepareSpreadsheet_(ss, '');
  Logger.log('Master template: ' + ss.getUrl() + '\nMake a copy of it for each Etsy store.');
  return ss.getUrl();
}

/** Registers the copy described in NEW_STORE. Re-running updates name/access, nothing else. */
function registerNewStore() {
  assertOwnerCaller_();
  var r = registerStore_(NEW_STORE);
  Logger.log(JSON.stringify(r, null, 2));
  return r;
}

function updateStoreAccess() {
  assertOwnerCaller_();
  var store = getStore_(ACCESS_UPDATE.storeId);
  if (!store) throw new Error('Unknown storeId.');
  var admins = normEmails_(ACCESS_UPDATE.admins);
  if (!admins.length) throw new Error('At least one administrator is required.');
  var team = normEmails_(ACCESS_UPDATE.team);
  store.members = membersFromLists_(admins, team, store.members);
  saveStore_(store);
  Logger.log('Access updated for ' + store.name + ': ' + admins.length + ' admin(s), ' + team.length + ' team member(s).');
}

/* ===================== internals ===================== */

function normEmails_(arr) {
  var seen = {};
  var out = [];
  (arr || []).forEach(function (x) {
    var e = String(x || '').trim().toLowerCase();
    if (!e) return;
    if (!EMAIL_RE_.test(e)) throw new Error('Invalid email address: ' + x);
    if (e === 'owner@example.com' || e === 'designer@example.com') throw new Error('Replace the example email addresses in NEW_STORE first.');
    if (!seen[e]) { seen[e] = true; out.push(e); }
  });
  return out;
}

function registerStore_(cfg) {
  if (scriptProp_('SIGNING_KEY').length < 32) throw new Error('Run initializeProject() first.');
  var ssId = String(cfg.spreadsheetId || '').trim();
  if (!/^[A-Za-z0-9_-]{20,}$/.test(ssId)) throw new Error('NEW_STORE.spreadsheetId is missing or malformed.');
  var nm = normalizeStoreName_(cfg.name);
  if (!nm.ok) throw new Error(nm.message);
  var admins = normEmails_(cfg.admins);
  var team = normEmails_(cfg.team);
  if (!admins.length) throw new Error('At least one administrator email is required.');
  if (ssId === scriptProp_('MASTER_TEMPLATE_ID')) {
    throw new Error('That is the master template itself. Make a copy of it and register the copy.');
  }

  var ss = SpreadsheetApp.openById(ssId);
  var webUrl = scriptProp_('WEBAPP_URL');
  var existing = findStoreBySpreadsheetId_(ssId);

  if (existing) {
    // Idempotent re-run: never clears data. Only name/access are updated and automation repaired.
    existing.name = nm.name;
    existing.members = membersFromLists_(admins, team, existing.members);
    saveStore_(existing);
    prepareSpreadsheet_(ss, existing.id);
    installEditTrigger_(ssId);
    try { ss.rename(nm.name); } catch (e) { /* optional */ }
    return { created: false, storeId: existing.id, name: nm.name, url: webUrl ? webUrl + '?store=' + existing.id : '(set WEBAPP_URL)' };
  }

  var meta = ss.getSheetByName(META_SHEET_);
  var stamp = meta ? String(meta.getRange(1, 2).getValue()) : '';
  var sheet = ss.getSheetByName(LISTINGS_SHEET_);
  var hasData = !!sheet && sheet.getLastRow() > 1;
  var fromOtherStore = !!stamp && !!getStore_(stamp);
  if (hasData || fromOtherStore) {
    if (cfg.existingData !== 'EMPTY' && cfg.existingData !== 'PRESERVE') {
      throw new Error('This spreadsheet already contains listing data' + (fromOtherStore ? ' copied from another store' : '') +
        '. Set NEW_STORE.existingData to "EMPTY" (start this store with no listings) or "PRESERVE" (keep the rows). Nothing was changed.');
    }
    if (cfg.existingData === 'EMPTY' && sheet) {
      var max = sheet.getMaxRows();
      if (max > 1) {
        sheet.getRange(2, 1, max - 1, NCOLS_).clearDataValidations();
        sheet.getRange(2, 1, max - 1, NCOLS_).clearContent();
      }
    }
  }

  var id;
  do { id = Utilities.getUuid().replace(/-/g, '').slice(0, 12); } while (getStore_(id));
  var store = { id: id, spreadsheetId: ssId, name: nm.name, members: membersFromLists_(admins, team, []) };
  var stamp0 = new Date().toISOString();
  store.members.forEach(function (m) { m.added = stamp0; m.by = scriptProp_('OWNER_EMAIL').toLowerCase(); });
  saveStore_(store);                   // registry first, so a failed re-run repairs instead of duplicating
  prepareSpreadsheet_(ss, id);
  installEditTrigger_(ssId);
  try { ss.rename(nm.name); } catch (e2) { /* optional */ }
  return { created: true, storeId: id, name: nm.name, url: webUrl ? webUrl + '?store=' + id : '(set WEBAPP_URL)' };
}

/** Builds / repairs the Listings sheet and hidden _Meta stamp. Idempotent. */
function prepareSpreadsheet_(ss, storeId) {
  ss.setSpreadsheetTimeZone(DISPLAY_TZ_);
  var sheet = ss.getSheetByName(LISTINGS_SHEET_);
  if (!sheet) sheet = ss.insertSheet(LISTINGS_SHEET_, 0);

  var headers = ['Etsy Listing Link', 'Listing ID', 'Open Listing', 'Notes', 'Images Done', 'Date Added', 'Date Completed',
    'Record ID', 'Seq', 'Status', 'Reason', 'Original Record ID', 'Version', 'Updated'];
  sheet.getRange(1, 1, 1, NCOLS_).setValues([headers]);

  var maxRows = Math.max(sheet.getMaxRows(), 500);
  if (sheet.getMaxRows() < maxRows) sheet.insertRowsAfter(sheet.getMaxRows(), maxRows - sheet.getMaxRows());
  var maxCols = sheet.getMaxColumns();
  if (maxCols < NCOLS_) sheet.insertColumnsAfter(maxCols, NCOLS_ - maxCols);

  sheet.getRange(1, 1, 1, 7).setBackground('#203B30').setFontColor('#FFFFFF').setFontWeight('bold').setVerticalAlignment('middle');
  sheet.getRange(1, 8, 1, 7).setBackground('#E0E5E1').setFontColor('#728579').setFontWeight('bold');
  sheet.setFrozenRows(1);
  sheet.setRowHeight(1, 34);
  sheet.setRowHeights(2, maxRows - 1, 30);
  sheet.getRange(2, 1, maxRows - 1, 7).setVerticalAlignment('middle').setFontColor('#34423A');
  sheet.getRange(2, 1, maxRows - 1, 7).setBorder(false, false, true, false, false, false, '#E0E5E1', SpreadsheetApp.BorderStyle.SOLID);
  sheet.setHiddenGridlines(true);

  // Plain-text columns: a pasted "=cmd()" or "+1" is stored as text and can never run as a formula.
  sheet.getRange(2, 1, maxRows - 1, 2).setNumberFormat('@');
  sheet.getRange(2, 4, maxRows - 1, 1).setNumberFormat('@').setWrap(true);
  sheet.getRange(2, 6, maxRows - 1, 2).setNumberFormat('dd mmm yyyy, hh:mm:ss AM/PM');
  sheet.getRange(2, 12, maxRows - 1, 1).setNumberFormat('@');

  var widths = [340, 130, 80, 260, 120, 190, 190];
  widths.forEach(function (w, i) { sheet.setColumnWidth(i + 1, w); });
  sheet.getRange(2, 3, maxRows - 1, 1).setHorizontalAlignment('center');
  sheet.getRange(2, 5, maxRows - 1, 1).setHorizontalAlignment('center');
  sheet.hideColumns(8, 7);

  var rng = sheet.getRange(2, 1, maxRows - 1, 7);
  var red = SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($J2<>"",$J2<>"VALID")').setBackground('#FBDCDC').setRanges([rng]).build();
  var green = SpreadsheetApp.newConditionalFormatRule()
    .whenFormulaSatisfied('=AND($J2="VALID",$E2=TRUE)').setBackground('#E1F2E7').setRanges([rng]).build();
  sheet.setConditionalFormatRules([red, green]); // replaces the list: never accumulates duplicates

  // Protect automatic fields (owner only). Real protection, not just hidden columns.
  sheet.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(function (p) {
    if (p.getDescription() === PROTECT_DESC_) p.remove();
  });
  var me = Session.getEffectiveUser();
  [sheet.getRange(1, 1, 1, NCOLS_), sheet.getRange(1, 2, maxRows, 2), sheet.getRange(1, 6, maxRows, 9)].forEach(function (r) {
    var prot = r.protect().setDescription(PROTECT_DESC_);
    prot.addEditor(me);
    prot.removeEditors(prot.getEditors().filter(function (u) { return u.getEmail() !== me.getEmail(); }));
    if (prot.canDomainEdit()) prot.setDomainEdit(false);
  });

  // Hidden identity stamp: ties this spreadsheet to exactly one registered store.
  var meta = ss.getSheetByName(META_SHEET_) || ss.insertSheet(META_SHEET_);
  meta.getRange(1, 1, 3, 2).setValues([
    ['storeId', storeId || ''],
    ['spreadsheetId', storeId ? ss.getId() : ''],
    ['note', 'Managed by Listing Studio. Do not edit.']
  ]);
  meta.hideSheet();

  var first = ss.getSheets()[0];
  if (first && first.getName() !== LISTINGS_SHEET_) ss.setActiveSheet(sheet);
  var stray = ss.getSheetByName('Sheet1');
  if (stray && stray.getLastRow() === 0 && ss.getSheets().length > 2) ss.deleteSheet(stray);
}

/** One installable edit trigger per store spreadsheet. */
function installEditTrigger_(spreadsheetId) {
  var exists = ScriptApp.getProjectTriggers().some(function (t) {
    return t.getHandlerFunction() === 'onSheetEdit' &&
      t.getTriggerSourceId() === spreadsheetId &&
      t.getEventType() === ScriptApp.EventType.ON_EDIT;
  });
  if (!exists) ScriptApp.newTrigger('onSheetEdit').forSpreadsheet(spreadsheetId).onEdit().create();
}
